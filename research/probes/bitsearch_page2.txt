    document.addEventListener('DOMContentLoaded', function() {
        const sortDropdown = document.getElementById('sort-dropdown');
        const sortOrder = document.getElementById('sort-order');
        const categoryDropdown = document.getElementById('category-dropdown');
        
        // Handle sort dropdown change
        if (sortDropdown) {
            sortDropdown.addEventListener('change', function() {
                const url = new URL(window.location);
                url.searchParams.set('sortBy', this.value);
                // Reset to page 1 when changing sort
                url.searchParams.set('page', '1');
                window.location.href = url.toString();
            });
        }
        
        // Handle sort order toggle
        if (sortOrder) {
            sortOrder.addEventListener('click', function() {
                const currentOrder = this.dataset.order;
                const newOrder = currentOrder === 'asc' ? 'desc' : 'asc';
                
                const url = new URL(window.location);
                url.searchParams.set('order', newOrder);
                // Reset to page 1 when changing sort order
                url.searchParams.set('page', '1');
                window.location.href = url.toString();
            });
        }
        
        // Handle category dropdown change
        if (categoryDropdown) {
            categoryDropdown.addEventListener('change', function() {
                const url = new URL(window.location);
                const value = this.value;
                
                // Clear existing category parameters
                url.searchParams.delete('category');
                url.searchParams.delete('subCategory');
                
                if (value) {
                    if (value.includes('-')) {
                        // Subcategory format: "6-4" (category 6, subcategory 4)
                        const [category, subCategory] = value.split('-');
                        url.searchParams.set('category', category);
                        url.searchParams.set('subCategory', subCategory);
                    } else {
                        // Main category only: "6"
                        url.searchParams.set('category', value);
                    }
                }
                
                // Reset to page 1 when changing category
                url.searchParams.set('page', '1');
                window.location.href = url.toString();
            });
        }

        // Impression tracking for search results
        (function() {
            var container = document.querySelector('[data-impression-ids]');
            if (!container) return;
            try {
                var ids = JSON.parse(container.getAttribute('data-impression-ids') || '[]');
                if (ids.length) {
                    fetch('/ping', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({ action: 'impressions', ids: ids })
                    }).catch(function() {});
                }
            } catch(e) {}
        })();
    });

                            if (data.suggestions && data.suggestions.length > 0) {

                                    `<div class="suggestion-item" data-index="${index}" onclick="selectSuggestion('${suggestion.replace(/'/g, '&#39;')}')">
                                        <i class="fas fa-search suggestion-icon"></i>
                                        <span>${suggestion}</span>
                                    </div>`
                                ).join('');
                                autocompleteResults.innerHTML = html;
                                showAutocomplete();
                            } else {
                                hideAutocomplete();
                            }
                        })
                        .catch(() => {
                            hideAutocomplete();
                        });
                }, 250);
            });

            // Keyboard navigation
            searchInput.addEventListener('keydown', function (e) {
                const suggestions = autocompleteResults.querySelectorAll('.suggestion-item');

                if (suggestions.length === 0) return;

                switch (e.key) {
                    case 'ArrowDown':
                        e.preventDefault();
                        highlightedIndex = Math.min(highlightedIndex + 1, suggestions.length - 1);
                        updateHighlight(suggestions);
                        break;
                    case 'ArrowUp':
                        e.preventDefault();
                        highlightedIndex = Math.max(highlightedIndex - 1, -1);
                        updateHighlight(suggestions);
                        break;
                    case 'Enter':
                        e.preventDefault();
                        if (highlightedIndex >= 0) {
                            const selectedSuggestion = suggestions[highlightedIndex].querySelector('span').textContent;
                            selectSuggestion(selectedSuggestion);
                        } else {
                            this.form.submit();
                        }
                        break;
                    case 'Escape':
                        hideAutocomplete();
                        highlightedIndex = -1;
                        break;
                }
            });

            // Hide suggestions when clicking outside
            document.addEventListener('click', function (event) {
                if (!searchInput.contains(event.target) && !autocompleteResults.contains(event.target)) {
                    hideAutocomplete();
                    highlightedIndex = -1;
                }
            });
        }

        // Smooth animation helper functions
        function showAutocomplete() {
            autocompleteResults.classList.remove('hidden');
            // Force a reflow to ensure the element is rendered before transition
            autocompleteResults.offsetHeight;
        }

        function hideAutocomplete() {
            autocompleteResults.classList.add('hidden');
        }

        function updateHighlight(suggestions) {
            suggestions.forEach((item, index) => {
                if (index === highlightedIndex) {
                    item.classList.add('highlighted');
                    // Auto-scroll to keep highlighted item visible
                    scrollToHighlighted(item);
                } else {
                    item.classList.remove('highlighted');
                }
            });
        }

        function scrollToHighlighted(highlightedItem) {
            const container = autocompleteResults;
            const containerHeight = container.clientHeight;
            const containerScrollTop = container.scrollTop;
            const itemTop = highlightedItem.offsetTop;
            const itemHeight = highlightedItem.offsetHeight;
            const itemBottom = itemTop + itemHeight;

            // Check if item is above the visible area
            if (itemTop < containerScrollTop) {
                container.scrollTo({
                    top: itemTop - 8, // Small offset for better visual spacing
                    behavior: 'smooth'
                });
            }
            // Check if item is below the visible area
            else if (itemBottom > containerScrollTop + containerHeight) {
                container.scrollTo({
                    top: itemBottom - containerHeight + 8, // Small offset for better visual spacing
                    behavior: 'smooth'
                });
            }
        }

        function selectSuggestion(suggestion) {
            searchInput.value = suggestion;
            hideAutocomplete();
            searchInput.form.submit();
        }

        // User menu dropdown functionality
        const userMenuButton = document.getElementById('userMenuButton');
        const userDropdown = document.getElementById('userDropdown');

        if (userMenuButton && userDropdown) {
            userMenuButton.addEventListener('click', function (e) {
                e.stopPropagation();
                userDropdown.classList.toggle('hidden');
            });

            // Close dropdown when clicking outside
            document.addEventListener('click', function (event) {
                if (!userMenuButton.contains(event.target) && !userDropdown.contains(event.target)) {
                    userDropdown.classList.add('hidden');
                }
            });
        }

        // Logout handlers
        const logoutBtn = document.getElementById('logoutBtn');
        const mobileLogoutBtn = document.getElementById('mobileLogoutBtn');

        async function handleLogout() {
            try {
                await fetch('/api/auth/logout', {
                    method: 'POST',
                    credentials: 'include' // Important: sends cookies
                });
            } catch (error) {
                console.error('Logout error:', error);
            }

            // Reload page to update UI
            window.location.href = '/';
        }

        if (logoutBtn) {
            logoutBtn.addEventListener('click', handleLogout);
        }

        if (mobileLogoutBtn) {
            mobileLogoutBtn.addEventListener('click', handleLogout);
        }
